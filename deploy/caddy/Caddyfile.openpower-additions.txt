# Managed for PalisBot + OpenPower.
#   - Admin API is fully disabled (no local attack surface, nothing to bind).
#   - PalisBot's own site blocks below declare an explicit "http://...:8080"
#     scheme+port, which Caddy's automatic HTTPS always leaves alone
#     regardless of the global auto_https setting -- so removing the blanket
#     "auto_https off" here does not change anything about how PalisBot is
#     served (still loopback-only, still fronted solely by the Cloudflare
#     Tunnel, still no certificate requested for it).
#   - openpower.one is the one site with no explicit scheme, so it's the one
#     site Caddy manages automatic HTTPS (Let's Encrypt) for -- see its block
#     below. It is also the only site here bound to a public interface.
{
	admin off
	servers {
		max_header_size 16KB
		# Traffic arrives from cloudflared over loopback, so the direct peer is
		# always 127.0.0.1. Without these, every access-log line and every
		# CrowdSec/fail2ban decision would see the tunnel instead of the real
		# client, making the bouncers both useless and dangerous (they would
		# ban loopback). CF-Connecting-IP is set by Cloudflare and is only
		# trusted because the sole path to this listener is the tunnel.
		trusted_proxies static 127.0.0.1/32 ::1
		client_ip_headers CF-Connecting-IP
	}
}

# www redirects to the apex: the oauth nonce cookie is host-only and
# discord.redirect_uri is fixed at the apex, so serving both hosts broke sign-in.
http://www.palisbot.com:8080 {
	bind 127.0.0.1
	redir https://palisbot.com{uri} permanent
}

http://127.0.0.1:8080, http://palisbot.com:8080 {
	bind 127.0.0.1

	# Default access log fields (method, uri, status, duration, size) only.
	# Headers, cookies, and request/response bodies are never logged.
	# Application routes must never place secrets in query strings.
	log {
		output file /var/log/caddy/access.log {
			roll_size 20mb
			roll_keep 10
			roll_keep_for 720h
		}
		format json
		level INFO
	}

	request_body {
		max_size 1MB
	}

	# Security headers are set HERE ONLY -- Next.js sets none, so there is
	# exactly one source and no conflicting duplicates. The CSP is built
	# around what the app actually needs:
	#   script-src 'self'          Next.js serves its own bundles; no CDN,
	#                              no inline <script> (JSON-LD is a
	#                              non-executable ld+json block), no eval.
	#   style-src  'unsafe-inline' required: Next.js inlines critical CSS
	#                              and React sets style attributes.
	#   connect-src 'self'         the browser only ever calls same-origin
	#                              /api; it never reaches Discord directly.
	#   img-src ... data:          inline placeholders/SVG favicon.
	#   form-action 'self'         login is a navigation to /api/..., which
	#                              is same-origin; Discord's consent screen
	#                              is reached by redirect, not form post.
	#   frame-ancestors 'none'     nothing may frame PalisBot.
	#   frame-src 'none'           PalisBot frames nothing (no Discord
	#                              embedding feature exists).
	header {
		X-Content-Type-Options nosniff
		Strict-Transport-Security "max-age=15552000"
		Referrer-Policy strict-origin-when-cross-origin
		Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=(), interest-cohort=()"
		-Server
		-X-Powered-By
	}

	header {
		defer
		-Via
		-X-Nextjs-Cache
		-X-Nextjs-Prerender
		-X-Nextjs-Stale-Time
		-X-Nextjs-Matched-Path
		-Vary
	}

	# Private routes must never be cached by an intermediary, and must
	# never be indexed even if a crawler reaches them directly.
	#
	# `defer` is essential: without it these are written before the
	# upstream responds, and Next.js's own Cache-Control (s-maxage for the
	# prerendered shell) is then *appended*, leaving two contradictory
	# Cache-Control headers. Deferring replaces the upstream value instead.
	@private path /dashboard /dashboard/* /login /auth-status
	header @private {
		defer
		Cache-Control "no-store, must-revalidate"
		X-Robots-Tag "noindex, nofollow"
	}

	# PalisBot API -> FastAPI over its existing Unix socket.
	# flush_interval -1 disables response buffering so future Server
	# Builder SSE/streaming endpoints reach the browser incrementally
	# instead of being held until the response completes.
	@probe {
		path /.env /.env.* /.git /.git/* /.svn/* /.hg/* /.aws/* /.ssh/* /.DS_Store
		path /backup.zip /backup.tar.gz /dump.sql /database.sql /db.sql
		path /phpmyadmin* /pma* /myadmin* /adminer* /wp-admin* /wp-login.php /wp-content/*
		path /xmlrpc.php /server-status /server-info /debug /config.json /configuration.php
		path /vendor/* /composer.json /composer.lock /.well-known/security.txt
		path /actuator* /solr/* /jenkins/* /_ignition/* /cgi-bin/*
	}
	handle @probe {
		respond 404
	}

	# Liveness/readiness are for systemd and local checks only. Served
	# publicly they confirm the stack and announce when PostgreSQL or
	# Valkey is down, which is reconnaissance and nothing else.
	@internal_api path /api/health /api/ready
	handle @internal_api {
		respond 404
	}

	handle /api/* {
		header {
			Content-Security-Policy "default-src 'none'; frame-ancestors 'none'; base-uri 'none'"
		}
		uri strip_prefix /api
		reverse_proxy unix//srv/palisbot/run/api.sock {
			flush_interval -1
		}
	}

	# Everything else -> Next.js standalone server on loopback.
	# Next.js serves its own /_next/static assets with correct immutable
	# caching, so there is no separate static handler to keep in sync.
	handle {
		encode zstd gzip
		# Two Next.js instances so a deploy can restart them one at a time and
		# a crashed process stops receiving traffic instead of returning 502.
		# Caddy already forwards Host and proto, so no header_up is needed.
		reverse_proxy 127.0.0.1:3000 127.0.0.1:3002 {
			lb_policy least_conn
			health_uri /healthz
			health_interval 10s
			health_timeout 3s
			fail_duration 15s
		}
	}
}

# Static assets only. Nothing dynamic and nothing authenticated is served here,
# so this hostname can be cached hard at the edge with no risk of storing a
# private response. Anything else answers 404 rather than falling through.

	bind 127.0.0.1

		output file /var/log/caddy/cdn.log {
			roll_size 20mb
			roll_keep 5
			roll_keep_for 720h
		}
		format json
	}
	header {
		X-Content-Type-Options nosniff
		Strict-Transport-Security "max-age=15552000"
		Access-Control-Allow-Origin "https://palisbot.com"
		X-Robots-Tag "noindex, nofollow"
		-Server
	}
	@assets path /_next/static/* /fonts/* /images/* /logos/* /servers/* /favicon.ico /favicon.svg /apple-touch-icon.png
	handle @assets {
		encode zstd gzip
		reverse_proxy 127.0.0.1:3000 127.0.0.1:3002 {
			lb_policy least_conn
			health_uri /healthz
			health_interval 10s
		}
	}
	handle {
		respond 404
	}
}
